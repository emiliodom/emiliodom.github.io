    function sleep(ms) {
      return new Promise((resolve) => setTimeout(resolve, ms));
    }

    function createTestRuntime() {
      const results = [];

      function test(name, fn) {
        try {
          fn();
          results.push({ name, ok: true });
        } catch (e) {
          results.push({ name, ok: false, error: e && e.message ? e.message : String(e) });
        }
      }

      function expect(received) {
        return {
          toBe(expected) {
            if (received !== expected) throw new Error(`Expected ${JSON.stringify(expected)}, got ${JSON.stringify(received)}`);
          },
          toBeCloseTo(expected, epsilon = 1e-9) {
            if (Math.abs(received - expected) > epsilon) throw new Error(`Expected ~${expected}, got ${received}`);
          },
        };
      }

      return { test, expect, results };
    }

    async function runJavaScriptChallenge(challenge, code) {
      const logs = [];
      const exportsObj = {};
      const consoleProxy = {
        log: (...args) => logs.push(args.map(String).join(' ')),
        warn: (...args) => logs.push('[warn] ' + args.map(String).join(' ')),
        error: (...args) => logs.push('[error] ' + args.map(String).join(' ')),
      };

      function require() {
        throw new Error('require() is not available in this sandbox');
      }

      const runtime = createTestRuntime();

      try {
        const fn = new Function('exports', 'console', 'require', 'sleep', 'test', 'expect', `"use strict";\n${code}\n;return exports;`);
        fn(exportsObj, consoleProxy, require, sleep, runtime.test, runtime.expect);
      } catch (e) {
        return {
          ok: false,
          lines: [{ ok: false, label: 'fail', text: 'Code execution error: ' + (e && e.message ? e.message : String(e)) }],
          logs,
        };
      }

      try {
        const maybePromise = challenge.tests(exportsObj);
        if (maybePromise && typeof maybePromise.then === 'function') {
          await maybePromise;
        }
      } catch (e) {
        // keep results, but mark failure
        runtime.results.push({ name: 'test runtime', ok: false, error: e && e.message ? e.message : String(e) });
      }

      return { ok: runtime.results.every(r => r.ok), testResults: runtime.results, logs };
    }

    function runPatternChecks(challenge, code) {
      const checks = challenge.checks || [];
      const lines = checks.map(chk => {
        try {
          const ok = chk.re ? chk.re.test(code) : chk.fn ? !!chk.fn(code) : false;
          return { ok, label: ok ? 'ok' : 'fail', text: chk.desc };
        } catch (e) {
          return { ok: false, label: 'fail', text: chk.desc + ' (check error)' };
        }
      });
      return { ok: lines.every(l => l.ok), lines };
    }
